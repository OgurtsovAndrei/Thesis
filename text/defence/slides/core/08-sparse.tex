\begin{frame}{Sparse Tail: Truncate $t$ Low Bits}
    \begin{columns}[T,onlytextwidth]
        \column{0.50\textwidth}
        \textbf{Random hash (SODA).}
        \emph{Phantoms} ($y \in U \setminus S$ with $h(y) \in h(S)$) ---
        $2^t$ per key, scattered uniformly across $U$.

        \medskip

        \textbf{Truncation hash} $h(x) = \lfloor x / 2^t \rfloor$.
        The $2^t$ phantoms of a key form a single contiguous block next to it.

        \column{0.48\textwidth}
        \centering
        \includegraphics[width=\linewidth, keepaspectratio]{../../figures/phantom_comparison}
    \end{columns}

    \begin{center}
        \fbox{\begin{minipage}{0.95\linewidth}
                  \centering
                  $\;\text{BPK:}\;\log_2(\mathcal{L}/\varepsilon)
                    \;\to\; \log_2(1/\varepsilon)\;$
        \end{minipage}}
    \end{center}

    \centering

    {
        \footnotesize\emph{Cost:} all phantoms are near keys. Requests close to the real key will cause False Positives.
        \\ On dense data all responses will be `non-empty` and FPR $\to 1$ --- so truncation only on \textbf{sparse} regions
    }
\end{frame}
