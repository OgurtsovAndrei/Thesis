\begin{frame}{Two Modes: Dense Clusters and Sparse Tails}
    \ourbadge
    \begin{columns}[T,onlytextwidth]
        \column{0.48\textwidth}
        \textbf{Dense cluster}
        \smallskip
        \begin{itemize}\setlength{\itemsep}{0.35em}
            \item ARE (Goswami): $\;n\log_2(\mathcal{L}/\varepsilon)\;$ bits
                  --- \textbf{independent of $U$}
            \item ERE: $\;n\log_2(U'/n) + O(n)\;$ bits
                  --- \textbf{shrinks with $U'$}
        \end{itemize}
        \medskip
        \begin{center}
            \fbox{Narrow enough cluster $\Longrightarrow$ exact mode, $\mathrm{FPR} = 0$}
        \end{center}

        \column{0.48\textwidth}
        \textbf{Sparse tail} (wide $U'$, few keys)
        \smallskip
        \begin{center}
            \includegraphics[width=0.92\linewidth, keepaspectratio]{../../figures/phantom_comparison_slide}
        \end{center}
        \begin{itemize}\setlength{\itemsep}{0.2em}
            \item Regular hash:\quad $\log_2(\mathcal{L}/\varepsilon)$ BPK
            \item Truncation:\quad $\log_2(1/\varepsilon)$ BPK
        \end{itemize}
        $\log_2(\text{Range Length})$ bits improvement
    \end{columns}
\end{frame}
